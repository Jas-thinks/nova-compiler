cmake_minimum_required(VERSION 3.16)
project(nova_compiler LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wno-unused-parameter)
endif()
option(NOVA_SANITIZE "Build with AddressSanitizer + UndefinedBehaviorSanitizer" OFF)
if(NOVA_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()
enable_testing()

file(GLOB_RECURSE NOVA_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
list(FILTER NOVA_SOURCES EXCLUDE REGEX ".*/novac\\.cpp$")
add_library(nova_core STATIC ${NOVA_SOURCES})
target_include_directories(nova_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

add_executable(novac src/tools/novac.cpp)
target_link_libraries(novac PRIVATE nova_core)

target_compile_definitions(nova_core PUBLIC NOVA_EXAMPLES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../examples")
add_executable(nova_tests tests/test_main.cpp tests/test_compiler.cpp tests/test_analysis.cpp)
target_compile_definitions(nova_tests PRIVATE NOVA_REGRESSIONS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/regressions")
target_link_libraries(nova_tests PRIVATE nova_core)
add_test(NAME nova_unit_tests COMMAND nova_tests)
